-- 2. FPS boost: remove visual-only nodes, NOT Workspace/Map, collisions,
--    triggers, player physics, NPCs, fruit handles or raid locations.
local function visualOnly(node)
    if node:IsA("Decal") or node:IsA("Texture") or node:IsA("SurfaceAppearance")
        or node:IsA("ParticleEmitter") or node:IsA("Trail") or node:IsA("Beam")
        or node:IsA("Fire") or node:IsA("Smoke") or node:IsA("Sparkles") then
        node:Destroy()
    elseif node:IsA("BasePart") then
        node.Material = Enum.Material.SmoothPlastic
        node.Reflectance = 0
        node.CastShadow = false
        -- "Delete Workspace" visuals without actually deleting physical objects:
        -- even collidable map parts stay in Workspace and keep collision/physics.
        local map = WS:FindFirstChild("Map")
        if CFG.HideMapVisuals and map and node.Anchored and node:IsDescendantOf(map) then
            node.LocalTransparencyModifier = 1
        end
    elseif node:IsA("PointLight") or node:IsA("SpotLight") or node:IsA("SurfaceLight") then
        node.Enabled = false
    end
end

-- 4. BNN's ItemReplicationService inventory and GetFruits price catalogue.
local ItemReplicationService = try(function() return require(RS:WaitForChild("ItemReplicationService", 15)) end)
local ItemConfig = try(function() return require(RS:WaitForChild("ItemConfig", 15)) end)
local PriceByStorageName = {}
local lastPriceFetch = 0
local lastInventoryCheck = 0
local cachedCheapCount = 0
local cachedCheapName = nil
local function refreshFruitPrices()
    if next(PriceByStorageName) and tick() - lastPriceFetch < 240 then return end
    local catalog = invoke("GetFruits", false)
    if type(catalog) ~= "table" then return end
    local updated = {}
    for _, fruit in pairs(catalog) do
        if type(fruit) == "table" and type(fruit.Name) == "string" and tonumber(fruit.Price) then
            updated[fruit.Name] = tonumber(fruit.Price)
        end
    end
    if next(updated) then PriceByStorageName = updated; lastPriceFetch = tick() end
end
local function priceFor(name)
    local direct = PriceByStorageName[name]
    if direct then return direct end
    local lowercase = string.lower(tostring(name or ""))
    for key, value in pairs(PriceByStorageName) do
        if string.lower(key) == lowercase then return value end
    end
end
local function cheapStoredFruit(force)
    if not CFG.GetCheapStoredFruit then return nil, 0 end
    if not force and tick() - lastInventoryCheck < 2 then return cachedCheapName, cachedCheapCount end
    lastInventoryCheck = tick()
    refreshFruitPrices()
    local best, bestPrice, amount = nil, math.huge, 0
    if not ItemReplicationService or not ItemConfig then return nil, 0 end
    try(function()
        local keys = ItemReplicationService.KEYS
        for _, entry in ItemReplicationService:GetItems(keys.QUANTITY) do
            if entry and tonumber(entry.Value) and entry.Value > 0 then
                local ok, meta = pcall(function() return ItemConfig.match(entry.ItemId):unwrap() end)
                if ok and meta and meta.Display and meta.Display.Category == "Blox Fruit" then
                    local name = (meta.Index and meta.Index.StorageKey) or meta.Display.Name
                    local price = priceFor(name)
                    -- < 2M: never consume unknown-price or expensive stored fruits.
                    if price and price < CFG.FruitPriceLimit then
                        amount += entry.Value
                        if price < bestPrice then best, bestPrice = name, price end
                    end
                end
            end
        end
    end)
    cachedCheapName, cachedCheapCount = best, amount
    return best, amount
end
local function worldFruit()
    local best, distance = nil, math.huge
    for _, item in ipairs(WS:GetChildren()) do
        if (item:IsA("Tool") or item:IsA("Model")) and item.Name:find("Fruit") then
            local handle = item:FindFirstChild("Handle")
            if handle and handle:IsA("BasePart") then
                local d = distFrom(handle.Position)
                if d < distance then distance, best = d, item end
            end
        end
    end
    return best
end

-- 5. Transport. Tween, short retarget cooldown, client-character noclip only
-- while moving. Workspace collisions remain untouched.
local moveTween, moveDestination, lastMove = nil, nil, 0
local noclip = false
local collisionCache = {}
local floatForce = nil
local function stopMove()
    if moveTween then try(function() moveTween:Cancel() end) end
    moveTween, moveDestination = nil, nil
    noclip = false
    if floatForce then try(function() floatForce:Destroy() end) end
    floatForce = nil
    for part, previous in pairs(collisionCache) do
        if part and part.Parent then try(function() part.CanCollide = previous end) end
        collisionCache[part] = nil
    end
end
RunService.Stepped:Connect(function()
    if not RUNNING or not noclip then return end
    local ch = LP.Character
    if ch then
        for _, part in ipairs(ch:GetDescendants()) do
            if part:IsA("BasePart") then
                if collisionCache[part] == nil then collisionCache[part] = part.CanCollide end
                part.CanCollide = false
            end
        end
    end
end)
local function toTarget(cf)
    if typeof(cf) ~= "CFrame" then return end
    local ch, hum, root = character()
    if not root or not hum or hum.Health <= 0 then return end
    local distance = (cf.Position - root.Position).Magnitude
    if distance < 7 then
        stopMove()
        root.CFrame = cf
        return
    end
    if moveDestination and (moveDestination.Position - cf.Position).Magnitude < 24 and moveTween then return end
    if tick() - lastMove < .13 then return end
    lastMove = tick()
    if moveTween then try(function() moveTween:Cancel() end) end
    moveDestination = cf
    noclip = true
    if not floatForce or not floatForce.Parent then
        floatForce = Instance.new("BodyVelocity")
        floatForce.Name = "BNN_RaidFloat"
        floatForce.MaxForce = Vector3.new(0, math.huge, 0)
        floatForce.Velocity = Vector3.zero
        floatForce.Parent = root
    end
    local speed = math.max(tonumber(CFG.TweenSpeed) or 310, 50)
    moveTween = TweenService:Create(root, TweenInfo.new(distance / speed, Enum.EasingStyle.Linear), { CFrame = cf })
    moveTween:Play()
end

function _FA_FastAttack:GetBladeHits(Character, Distance, IncludePlayers)
	local Position = Character:GetPivot().Position
	local BladeHits = {}
	Distance = Distance or _FA_AttackConfig.AttackDistance
	self.EnemyRootPart = nil

	local function ProcessTargets(Folder)
		for _, Enemy in ipairs(Folder:GetChildren()) do
			pcall(function()
				if Enemy ~= Character and self:IsEntityAlive(Enemy) then
					local BasePart = Enemy:FindFirstChild(_FA_AttackConfig.HitboxLimbs[math.random(#_FA_AttackConfig.HitboxLimbs)])
						or Enemy:FindFirstChild("HumanoidRootPart")
					if BasePart and (Position - BasePart.Position).Magnitude <= Distance then
						if not self.EnemyRootPart then
							self.EnemyRootPart = BasePart
						else
							table.insert(BladeHits, { Enemy, BasePart })
						end
					end
				end
			end)
		end
	end

	if _FA_AttackConfig.AttackMobs then
		pcall(ProcessTargets, _FA_Workspace:WaitForChild("Enemies"))
	end
	if IncludePlayers or _FA_AttackConfig.AttackPlayers then
		pcall(ProcessTargets, _FA_Workspace:WaitForChild("Characters"))
	end
	return BladeHits
end

function _FA_FastAttack:GetClosestEnemy(Character, Distance)
	self:GetBladeHits(Character, Distance, false)
	return self.EnemyRootPart
end

function _FA_FastAttack:GetCombo()

	Combo = Combo >= _FA_AttackConfig.MaxCombo and 1 or Combo + 1
	self.ComboDebounce = tick()
	self.M1Combo = Combo
	return Combo
end

function _FA_FastAttack:GetValidator2()
	if not self.ShootFunction then
		return nil
	end
	local v1 = debug.getupvalue(self.ShootFunction, 15)
	local v2 = debug.getupvalue(self.ShootFunction, 13)
	local v3 = debug.getupvalue(self.ShootFunction, 16)
	local v4 = debug.getupvalue(self.ShootFunction, 17)
	local v5 = debug.getupvalue(self.ShootFunction, 14)
	local v6 = debug.getupvalue(self.ShootFunction, 12)
	local v7 = debug.getupvalue(self.ShootFunction, 18)
	if not (v1 and v2 and v3 and v4 and v5 and v6 and v7) then
		return nil
	end
	local v8 = v6 * v2
	local v9 = (v5 * v2 + v6 * v1) % v3
	v9 = (v9 * v3 + v8) % v4
	v5 = math.floor(v9 / v3)
	v6 = v9 - v5 * v3
	v7 = v7 + 1
	debug.setupvalue(self.ShootFunction, 15, v1)
	debug.setupvalue(self.ShootFunction, 13, v2)
	debug.setupvalue(self.ShootFunction, 16, v3)
	debug.setupvalue(self.ShootFunction, 17, v4)
	debug.setupvalue(self.ShootFunction, 14, v5)
	debug.setupvalue(self.ShootFunction, 12, v6)
	debug.setupvalue(self.ShootFunction, 18, v7)
	return math.floor(v9 / v4 * 16777215), v7
end

function _FA_FastAttack:ShootInTarget(TargetPosition)
	local Character = _FA_Player.Character
	if not self:IsEntityAlive(Character) then return end
	local Equipped = Character:FindFirstChildOfClass("Tool")
	if not Equipped or Equipped.ToolTip ~= "Gun" then return end
	local Cooldown = Equipped:FindFirstChild("Cooldown") and Equipped.Cooldown.Value or 0.3
	if (tick() - self.ShootDebounce) < Cooldown then return end
	local ShootType = self.SpecialShoots[Equipped.Name] or "Normal"

	-- Skull Guitar needs a normal M1 input on some clients/executors.
	-- Try the real click path first; only use TAP remote if the click did not
	-- advance LocalTotalShots, avoiding an accidental double shot.
	if Equipped.Name == "Skull Guitar" then
		local BeforeShots = Equipped:GetAttribute("LocalTotalShots") or 0
		_FA_ClickM1()
		task.wait(0.025)
		local AfterShots = Equipped:GetAttribute("LocalTotalShots") or 0

		if AfterShots <= BeforeShots and Equipped:FindFirstChild("RemoteEvent") then
			Equipped:SetAttribute("LocalTotalShots", BeforeShots + 1)
			local Validator, Counter = self:GetValidator2()
			if Validator and Counter then
				_FA_GunValidator:FireServer(Validator, Counter)
			end
			Equipped.RemoteEvent:FireServer("TAP", TargetPosition)
		end

		self.ShootDebounce = tick()
		return
	end

	if ShootType == "Position" or (ShootType == "TAP" and Equipped:FindFirstChild("RemoteEvent")) then
		Equipped:SetAttribute("LocalTotalShots", (Equipped:GetAttribute("LocalTotalShots") or 0) + 1)
		local Validator, Counter = self:GetValidator2()
		if Validator and Counter then
			_FA_GunValidator:FireServer(Validator, Counter)
		end
		if ShootType == "TAP" then
			Equipped.RemoteEvent:FireServer("TAP", TargetPosition)
		else
			_FA_ShootGunEvent:FireServer(TargetPosition)
		end
		self.ShootDebounce = tick()
	else
		_FA_ClickM1()
		self.ShootDebounce = tick()
	end
end

function _FA_FastAttack:Attack()
	if not _FA_AttackConfig.AutoClickEnabled or tick() > self.ActiveUntil then return end
	if (tick() - self.Debounce) < _FA_AttackConfig.AttackCooldown then return end
	local Character = _FA_Player.Character
	if not Character or not self:IsEntityAlive(Character) then return end
	local Humanoid = Character:FindFirstChild("Humanoid")
	local Equipped = Character:FindFirstChildOfClass("Tool")
	if not Humanoid or not Equipped then return end
	local ToolTip = Equipped.ToolTip
	if not table.find({ "Melee", "Blox Fruit", "Sword", "Gun" }, ToolTip) then return end
	local Cooldown = Equipped:FindFirstChild("Cooldown") and Equipped.Cooldown.Value or _FA_AttackConfig.AttackCooldown
	if not self:CheckStun(Character, Humanoid, ToolTip) then return end
	local Combo = self:GetCombo()
	Cooldown = Cooldown + (Combo >= _FA_AttackConfig.MaxCombo and 0.05 or 0)
	self.Debounce = Combo >= _FA_AttackConfig.MaxCombo and ToolTip ~= "Gun" and (tick() + 0.05) or tick()
	if ToolTip == "Blox Fruit" and Equipped:FindFirstChild("LeftClickRemote") then
		self:UseFruitM1(Character, Equipped, Combo)
	elseif ToolTip == "Gun" then
		local Target = self:GetClosestEnemy(Character, math.max(120, _FA_AttackConfig.AttackDistance))
		if Target then self:ShootInTarget(Target.Position) end
	else
		self:UseNormalClick(Character, Humanoid, Cooldown)
	end
end

-- 7. Raid bring-mob: same-name, nearby mobs; only client-network owned
-- parts are moved. No anchoring or destruction of NPC physics objects.
local lastBring = 0
local function bringMobRaid(target)
    if not CFG.BringMob or tick() - lastBring < .14 or not alive(target) then return end
    local _, _, playerRoot = character()
    local targetRoot = target.HumanoidRootPart
    if not playerRoot or (playerRoot.Position - targetRoot.Position).Magnitude > 90 then return end
    lastBring = tick()
    if type(sethiddenproperty) == "function" then
        try(function() sethiddenproperty(LP, "SimulationRadius", math.huge) end)
    end
    local moved = 0
    local enemies = WS:FindFirstChild("Enemies")
    if not enemies then return end
    for _, mob in ipairs(enemies:GetChildren()) do
        if mob ~= target and mob.Name == target.Name and alive(mob) and moved < CFG.MaxMobsToBring then
            local root = mob.HumanoidRootPart
            if (root.Position - targetRoot.Position).Magnitude <= 200 then
                local owns = false
                if type(isnetworkowner) == "function" then
                    owns = try(function() return isnetworkowner(root) end) == true
                else
                    owns = true -- BNN fallback heuristic if executor has no owner API
                    local characters = WS:FindFirstChild("Characters")
                    if characters then
                        for _, other in ipairs(characters:GetChildren()) do
                            if other.Name ~= LP.Name and other:FindFirstChild("HumanoidRootPart")
                                and (other.HumanoidRootPart.Position - root.Position).Magnitude < 300 then
                                owns = false; break
                            end
                        end
                    end
                end
                if owns then
                    root.CanCollide = false
                    root.AssemblyLinearVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero
                    root.CFrame = targetRoot.CFrame * CFrame.new(0, 0, moved % 3)
                    moved += 1
                end
            end
        end
    end
end

local function queueAgain()
    local queue = (type(queue_on_teleport) == "function" and queue_on_teleport)
        or (type(syn) == "table" and type(syn.queue_on_teleport) == "function" and syn.queue_on_teleport)
    if not queue then return false, "executor lacks queue_on_teleport" end
    if type(ENV.__BNN_RAID_SCRIPT_BODY) ~= "string" then return false, "embedded source unavailable" end
    local persisted = {}
    for job in pairs(visited) do
        persisted[#persisted + 1] = job
        if #persisted >= 80 then break end
    end
    local cfgJSON = Http:JSONEncode(CFG)
    local idsJSON = Http:JSONEncode(persisted)
    local script = 'local __source=' .. string.format('%q', ENV.__BNN_RAID_SCRIPT_BODY) .. '\n'
        .. 'local e=getgenv()\ne.__BNN_RAID_RUNNING=nil\ne.__BNN_RAID_SCRIPT_BODY=__source\n'
        .. 'e.__BNN_RAID_CONFIG_JSON=' .. string.format('%q', cfgJSON) .. '\n'
        .. 'local list=game:GetService("HttpService"):JSONDecode(' .. string.format('%q', idsJSON) .. ')\n'
        .. 'e.__BNN_RAID_VISITED={}\nfor _,id in ipairs(list) do e.__BNN_RAID_VISITED[id]=true end\n'
        .. 'e.__BNN_RAID_HOPS=' .. tostring(HOPS + 1) .. '\nloadstring(__source)()'
    return pcall(queue, script)
end

local function requestServerPage(url, timeout)
    local req = (type(request) == "function" and request)
        or (type(http_request) == "function" and http_request)
        or (type(syn) == "table" and type(syn.request) == "function" and syn.request)
        or (type(http) == "table" and type(http.request) == "function" and http.request)
    -- Skider's request() / game:HttpGet() fallback, wrapped so the hop worker
    -- remains responsive when an executor or API request does not return.
    local op = {done = false}
    task.spawn(function()
        local ok, body, code, headers = pcall(function()
            if req then
                local response = req({Url = url, Method = "GET", Headers = {["Cache-Control"] = "no-cache"}})
                if type(response) ~= "table" then error("invalid HTTP response") end
                return response.Body, tonumber(response.StatusCode or response.Status) or 0, response.Headers
            end
            return game:HttpGet(url), 200, nil
        end)
        op.ok, op.body, op.code, op.headers, op.done = ok, body, code, headers, true
    end)
    local deadline = os.clock() + (timeout or 12)
    while RUNNING and not op.done and os.clock() < deadline do task.wait(.1) end
    if not op.done then return nil, "HTTP request timeout", 0 end
    if not op.ok then return nil, tostring(op.body), 0 end
    if op.code == 429 then
        local h = type(op.headers) == "table" and op.headers or {}
        local cooldown = math.max(30, tonumber(h["Retry-After"] or h["retry-after"]) or 30)
        return nil, "HTTP 429 rate limited", cooldown
    end
    if op.code ~= 200 and op.code ~= 0 then return nil, "HTTP " .. tostring(op.code), 0 end
    local valid, data = pcall(Http.JSONDecode, Http, op.body or "")
    if not valid or type(data) ~= "table" or type(data.data) ~= "table" then
        return nil, "Invalid public servers response", 0
    end
    return data, nil, 0
end

local function scanPublicServers()
    local now = os.clock()
    for id, expireAt in pairs(failedServers) do
        if expireAt <= now then failedServers[id] = nil end
    end
    local base = "https://games.roblox.com/v1/games/" .. tostring(game.PlaceId)
        .. "/servers/Public?sortOrder=Desc&excludeFullGames=true&limit=100"
    local cursor, cursors = nil, {}
    local pool, seen = {}, {}
    local maxPages = math.clamp(tonumber(CFG.ServerScanPages) or 20, 1, 20)
    for page = 1, maxPages do
        if not RUNNING then return pool, "Stopped", 0 end
        setStatus(string.format("Fruit hop: scanning page %d/%d", page, maxPages))
        local url = base .. (cursor and ("&cursor=" .. Http:UrlEncode(cursor)) or "")
        local data, err, cooldown
        for attempt = 1, 3 do
            data, err, cooldown = requestServerPage(url, 12)
            if data then break end
            if attempt < 3 then
                local delay = math.max(2 ^ attempt, cooldown or 0)
                setStatus(string.format("Fruit hop: %s, retry in %ds", tostring(err), delay))
                task.wait(delay)
            end
        end
        if not data then return pool, err, cooldown end
        for _, server in ipairs(data.data) do
            local id = server.id
            local count, capacity = tonumber(server.playing), tonumber(server.maxPlayers)
            if type(id) == "string" and id ~= "" and id ~= tostring(game.JobId)
                and not seen[id] and not visited[id] and not failedServers[id]
                and count and capacity and count > 0 and count < capacity then
                seen[id] = true
                pool[#pool + 1] = {id = id, playing = count, random = math.random()}
            end
        end
        if #pool >= 8 then break end
        local nextCursor = data.nextPageCursor
        if type(nextCursor) ~= "string" or nextCursor == "" or cursors[nextCursor] then break end
        cursors[nextCursor] = true
        cursor = nextCursor
        task.wait(.75)
    end
    table.sort(pool, function(a, b)
        if a.playing ~= b.playing then return a.playing > b.playing end
        return a.random < b.random
    end)
    return pool, nil, 0
end

local function tryTeleportServer(server)
    if not RUNNING then return false end
    teleportStarted, teleportFailure = false, nil
    -- Like Skider: call TeleportToPlaceInstance directly, not __ServerBrowser.
    setStatus("Fruit hop: joining server " .. server.id:sub(1, 8))
    local queued, queueError = queueAgain()
    if not queued then
        ERR = "Auto restart unavailable: " .. tostring(queueError)
        warn("[AUTO RAID / HOP] " .. ERR)
    end
    local op = {done = false}
    task.spawn(function()
        local ok, err = pcall(function()
            stopMove()
            TeleportService:TeleportToPlaceInstance(game.PlaceId, server.id, LP)
        end)
        op.done, op.ok, op.error = true, ok, err
    end)
    local deadline = os.clock() + 15
    local extended = false
    while RUNNING and not teleportFailure and os.clock() < deadline do
        if teleportStarted and not extended then
            extended = true
            deadline = os.clock() + 40
        end
        if op.done and not op.ok then teleportFailure = tostring(op.error); break end
        task.wait(.1)
    end
    -- Successful teleport leaves this client. Reaching here is a rejected/stalled hop.
    ERR = teleportFailure or "Teleport not confirmed (timeout)"
    failedServers[server.id] = os.clock() + 120
    warn("[AUTO RAID / HOP] " .. ERR)
    return false
end

local function hopServer()
    if hopping or os.clock() - lastHop < CFG.HopCooldown then return end
    hopping = true
    lastHop = os.clock()
    stopMove()
    task.spawn(function()
        local ok, reason = pcall(function()
            setStatus("Fruit hop: finding public servers")
            local pool, errorMessage, cooldown = scanPublicServers()
            if errorMessage then
                ERR = tostring(errorMessage)
                if #pool == 0 then
                    setStatus("Fruit hop: scan failed - " .. ERR)
                    if cooldown and cooldown > 0 then lastHop = os.clock() + cooldown - CFG.HopCooldown end
                    return
                end
            end
            if #pool == 0 then
                ERR = "No public servers with free slots"
                setStatus("Fruit hop: no eligible server")
                return
            end
            for i = 1, math.min(8, #pool) do
                if not RUNNING then return end
                if tryTeleportServer(pool[i]) then return end
                task.wait(1)
            end
            hopFailureCounter += 1
            setStatus("Fruit hop: all candidates failed; retrying")
        end)
        if not ok then
            ERR = tostring(reason)
            setStatus("Fruit hop error: " .. ERR)
            warn("[AUTO RAID / HOP] " .. ERR)
        end
        hopping = false
        lastHop = math.max(lastHop, os.clock())
    end)
end

			getgenv().DaBringMob = false
		end)
		return
	end
	local E = {}
	if not b:FindFirstChild("Ignored") then
		table.insert(E, b)
	end
	for y, y in pairs(game:GetService("Workspace").Enemies:GetChildren()) do
		if
			y ~= b
			and y.Name == b.Name
			and not y:FindFirstChild("Ignored")
			and (IsMobAlive(y))
			and (isnetworkowner2(y.HumanoidRootPart))
		then
			if (y.HumanoidRootPart.Position - l.Position).Magnitude <= 200 and #E < 1 then
				table.insert(E, y)
			end
		end
	end
	if
		l
		and (t.Character.HumanoidRootPart.Position - b.HumanoidRootPart.Position).Magnitude <= 50
		and (isnetworkowner2(t.Character.HumanoidRootPart))
	then
		for b, b in pairs(E) do
			sizepart(b)
			b.HumanoidRootPart.CFrame = l * CFrame.new(0, math.random(0, 2), math.random(0, 2))
			task.spawn(function()
				local E = b.Humanoid.Health
				task.wait(3.5)
				if b.Humanoid.Health == E and not b:FindFirstChild("Ignored") then
					b.HumanoidRootPart.CFrame = b.WorldPivot
					Instance.new("IntValue", b).Name = "Ignored"
					task.wait(0.3)
				end
			end)
			getgenv().DaBringMob = true
		end
	end
end
function GetLastRaidIsland()
	local b, E = 0
	for l, y in ipairs(wOrigin.Locations:GetChildren()) do
		if string.find(y.Name, "Island ") and t:DistanceFromCharacter(y.Position) < 3000 then
			l = tonumber((y.Name:gsub("Island ", "")))
			if b < l then
				b, E = l, y
			end
		end
	end
	return E
end
function CheckInRaid()
	for b, b in ipairs(wOrigin.Locations:GetChildren()) do
		if string.find(b.Name, "Island ") and t:DistanceFromCharacter(b.Position) < 3000 then
			return true
		end
	end
end

-- GUI refresh is isolated: sandboxed executors can deny access to particular
-- Instances ("lacking capability Plugin"). Such a denial must not kill status UI.
local function safeText(fn, fallback)
    local ok, value = pcall(fn)
    if ok and value ~= nil then return tostring(value) end
    return fallback or "?"
end
local function refreshStatusUI()
    local islandNum = safeText(function()
        local _, idx = newestRaidIsland()
        return idx and idx > 0 and idx or "-"
    end, "-")
    local cheapCount = safeText(function()
        local _, count = cheapStoredFruit(false)
        return count
    end, "?")
    local team = safeText(function() return LP.Team and LP.Team.Name end, "Choosing team")
    local chip = safeText(function() return hasItem("Special Microchip") and "YES" or "NO" end)
    local physicalFruit = safeText(function() return heldFruit() and "YES" or "NO" end)
    -- Never traverse LP.Data without pcall: some executors deny a protected Instance.
    local beli = safeText(function()
        local data = LP:FindFirstChild("Data")
        local money = data and data:FindFirstChild("Beli")
        return money and money.Value
    end)
    local level = safeText(function()
        local data = LP:FindFirstChild("Data")
        local lvl = data and data:FindFirstChild("Level")
        return lvl and lvl.Value
    end)
    head.Text = "AUTO RAID"
    stateLabel.Text = "STATUS  " .. PHASE
    infoLabel.Text = string.format("Sea %d  |  %s  |  Island %s  |  Hops %d", sea(), CFG.Raid, islandNum, HOPS)
    fruitLabel.Text = string.format("Chip: %s    Fruit held: %s    Cheap (<2M): %s    Beli: %s", chip, physicalFruit, cheapCount, beli)
    errorLabel.Text = ERR ~= "" and ("Last error: " .. ERR) or ("Aura: " .. safeText(function()
        local ch = LP.Character
        if not ch then return "checking" end
        if ch:FindFirstChild("HasBuso") then return "ON" end
        for _, obj in ipairs(ch:GetChildren()) do
            if string.find(obj.Name, "_BusoLayer1", 1, true) then return "ON" end
        end
        return "activating"
    end, "checking") .. " | Fast Attack | Bring Mob")
    raidLabel.Text = "Level " .. level
end

-- The original BNN Auto Turn On Buso routine: check HasBuso / _BusoLayer1
-- before invoking CommF_("Buso"). Do not toggle when already active.
local lastBusoCheck = -math.huge
local function ensureBuso()
    if not CFG.AutoBuso or tick() - lastBusoCheck < 2 then return end
    lastBusoCheck = tick()
    local ch = LP.Character
    if not ch then return end
    local active = ch:FindFirstChild("HasBuso") ~= nil
    if not active then
        for _, child in ipairs(ch:GetChildren()) do
            if string.find(child.Name, "_BusoLayer1", 1, true) then
                active = true
                break
            end
        end
    end
    if not active then invoke("Buso") end
end
