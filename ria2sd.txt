-- 1. ULTRA FPS BOOST & XÓA VFX ĐÁNH NHAU
local function StripVFX(v)
    pcall(function()
        if v:IsA("BasePart") then
            v.Material = Enum.Material.Plastic
            v.Reflectance = 0
            v.CastShadow = false
        elseif v:IsA("Decal") or v:IsA("Texture") or v:IsA("SurfaceAppearance") then
            v:Destroy()
        elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") or v:IsA("Beam") or v:IsA("Highlight") then
            v:Destroy()
        elseif v:IsA("Explosion") then
            v.Visible = false
            v.BlastPressure = 0
        end
    end)
end

task.spawn(function()
    pcall(function()
        settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 9e9
        Lighting.Brightness = 0
        for _, v in ipairs(Lighting:GetChildren()) do
            if v:IsA("PostEffect") or v:IsA("Sky") or v:IsA("Atmosphere") then v:Destroy() end
        end
        pcall(function() require(RS.Util.CameraShaker):Stop() end)
        workspace.ClientAnimatorThrottling = Enum.ClientAnimatorThrottlingMode.Enabled
        workspace.InterpolationThrottling = Enum.InterpolationThrottlingMode.Enabled
        for _, v in ipairs(workspace:GetDescendants()) do StripVFX(v) end
        workspace.DescendantAdded:Connect(StripVFX)
        wOrigin.ChildAdded:Connect(function(c)
            if c.Name == "DamageCounter" or c.Name == "Foam" or c:IsA("ParticleEmitter") or c:IsA("BasePart") and c.Name ~= "Island 1" and not string.find(c.Name, "Island") then
                task.defer(function() pcall(c.Destroy, c) end)
            end
        end)
    end)
end)

    p.Transparency, p.CanCollide, p.Anchored, p.Size = 1, false, true, Vector3.zero
    p.CFrame = CFrame.new(pos)

end

-- 2. TWEEN, NÉ CHIÊU BOSS & HỒI MÁU (Chuẩn vxeze (1).lua)
local CurrentTween, CurrentTargetMob, DodgeUntil, LowHealth = nil, nil, 0, false

local function StopTween()
    if CurrentTween then pcall(function() CurrentTween:Cancel() end) CurrentTween = nil end
end

workspace.Enemies.DescendantAdded:Connect(function(desc)
    if not getgenv().AutoRaid13k or not CurrentTargetMob then return end
    if desc.Name == "BodyGyro" or desc.Name == "BodyPosition" or string.find(desc.Name, "Blast") then
        if desc:IsDescendantOf(CurrentTargetMob) then
            DodgeUntil = tick() + 1.6
        end
    end
end)

RunService.Stepped:Connect(function()
    if not getgenv().AutoRaid13k then return end
    pcall(function()
        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not (hrp and hum) then return end

        local hpPct = hum.MaxHealth > 0 and (hum.Health / hum.MaxHealth) or 1
        if hpPct < 0.42 then LowHealth = true elseif hpPct > 0.78 then LowHealth = false end

        if not hrp:FindFirstChild("FloatForce") then
            local bv = Instance.new("BodyVelocity", hrp)
            bv.Name, bv.Velocity, bv.MaxForce, bv.P = "FloatForce", Vector3.zero, Vector3.new(1e5, 1e5, 1e5), 15000
        end
        hrp.AssemblyLinearVelocity = Vector3.zero
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") and part.CanCollide then part.CanCollide = false end
        end
    end)
end)

local function TweenTo(targetCF, speed)
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not (hrp and hum and hum.Health > 0) then return end
    if hum.Sit then hum.Sit = false hum.Jump = true end

    local dist = (hrp.Position - targetCF.Position).Magnitude
    if dist <= 18 then
        StopTween()
        hrp.CFrame = targetCF
        return
    end
    StopTween()
    if dist > 60 then
        hrp.CFrame = CFrame.new(hrp.Position.X, targetCF.Position.Y, hrp.Position.Z)
    end
    CurrentTween = TS:Create(hrp, TweenInfo.new(dist / (speed or 250), Enum.EasingStyle.Linear), { CFrame = targetCF })
    CurrentTween:Play()
end

-- 3. RÚT FRUIT < 1.5M BELI
local FruitPrices = {}
local function GetCheapFruit()
    if not next(FruitPrices) then
        local ok, data = pcall(function() return CommF:InvokeServer("GetFruits", false) end)
        if ok and type(data) == "table" then
            for _, f in pairs(data) do if f.Name and f.Price then FruitPrices[f.Name] = tonumber(f.Price) end end
        end
    end
    local best, minP = nil, math.huge
    pcall(function()
        local IRS = require(RS:WaitForChild("ItemReplicationService"))
        local IC = require(RS:WaitForChild("ItemConfig"))
        for _, item in IRS:GetItems(IRS.KEYS.QUANTITY) do
            if item.Value and item.Value > 0 then
                local ok, cfg = pcall(function() return IC.match(item.ItemId):unwrap() end)
                if ok and cfg and cfg.Display and cfg.Display.Category == "Blox Fruit" then
                    local key = (cfg.Index and cfg.Index.StorageKey) or cfg.Display.Name
                    local p = FruitPrices[key] or tonumber(cfg.Price) or 500000
                    if key and p < 1500000 and p < minP then best, minP = key, p end
                end
            end
        end
    end)
    if not best then
        pcall(function()
            for _, v in pairs(CommF:InvokeServer("getInventory") or {}) do
                if v.Type == "Blox Fruit" and v.Name then
                    local p = tonumber(v.Value) or FruitPrices[v.Name] or 500000
                    if p < 1500000 and p < minP then best, minP = v.Name, p end
                end
            end
        end)
    end
    return best
end

local function HasChip()
    return LP.Backpack:FindFirstChild("Special Microchip") or (LP.Character and LP.Character:FindFirstChild("Special Microchip"))
end

local function HasFruit()
    for _, c in ipairs({ LP.Backpack, LP.Character }) do
        if c then
            for _, t in ipairs(c:GetChildren()) do
                if t:IsA("Tool") and (string.find(t.Name, "Fruit") or t:FindFirstChild("EatRemote")) then return t end
            end
        end
    end
end

-- 4. SUPER FAST ATTACK & BRING MOB (Không dùng Kill Aura lỗi)
local Net = require(RS.Modules.Net)
local RegAttack = RS.Modules.Net:WaitForChild("RE/RegisterAttack")
local RegHit = Net:RemoteEvent("RegisterHit", true)

task.spawn(function()
    while task.wait(0.04) do
        if not getgenv().AutoRaid13k then break end
        if getgenv().IsFarmingRaidMob then
            pcall(function()
                _G.tapCooldown = 0
                local char = LP.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                if not hrp or (char:FindFirstChild("Stun") and char.Stun.Value ~= 0) then return end

                local hits = {}
                for _, mob in ipairs(workspace.Enemies:GetChildren()) do
                    local mHrp = mob:FindFirstChild("HumanoidRootPart")
                    local mHum = mob:FindFirstChildOfClass("Humanoid")
                    if mHrp and mHum and mHum.Health > 0 and (mHrp.Position - hrp.Position).Magnitude <= 80 then
                        table.insert(hits, { mob, mob:FindFirstChild("Head") or mHrp })
                    end
                end
                if #hits > 0 then
                    local mainPart = table.remove(hits, 1)[2]
                    RegAttack:FireServer(0)
                    RegHit:FireServer(mainPart, hits)
                end
            end)
        end
    end
end)

local function BringMobs(centerMob)
    pcall(sethiddenproperty, LP, "SimulationRadius", math.huge)
    local cHrp = centerMob and centerMob:FindFirstChild("HumanoidRootPart")
    if not cHrp then return end
    local targetCF = cHrp.CFrame

    for _, mob in ipairs(workspace.Enemies:GetChildren()) do
        local mHrp = mob:FindFirstChild("HumanoidRootPart")
        local mHum = mob:FindFirstChildOfClass("Humanoid")
        if mHrp and mHum and mHum.Health > 0 and (mHrp.Position - targetCF.Position).Magnitude <= 350 then
            pcall(function()
                for _, p in ipairs(mob:GetChildren()) do
                    if p:IsA("BasePart") then p.CanCollide = false end
                end
                mHum.WalkSpeed = 0
                mHum.JumpPower = 0
                if mob ~= centerMob then
                    mHrp.CFrame = targetCF
                end
                mHrp.AssemblyLinearVelocity = Vector3.zero
            end)
        end
    end
end

local LastBuffTime = 0
local function EquipAndBuff()
    local char = LP.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not (char and hum and hum.Health > 0) then return end

    local tool = char:FindFirstChildOfClass("Tool")
    if not (tool and tool.ToolTip == "Melee") then
        for _, t in ipairs(LP.Backpack:GetChildren()) do
            if t:IsA("Tool") and t.ToolTip == "Melee" then hum:EquipTool(t) break end
        end
    end

    if tick() - LastBuffTime > 1.5 then
        LastBuffTime = tick()
        if not char:FindFirstChild("HasBuso") then pcall(function() CommF:InvokeServer("Buso") end) end
        pcall(function() CommE:FireServer("ActivateAbility") end) -- Bật Tộc V3 (Water Body giảm 80% sát thương)
        local awk = LP.Backpack:FindFirstChild("Awakening") or char:FindFirstChild("Awakening")
        if awk and awk:FindFirstChild("RemoteFunction") then
            pcall(function() awk.RemoteFunction:InvokeServer(true) end)
        end
    end
end

-- 5. NHẬN DIỆN RAID & VÒNG LẶP CHÍNH
local function GetRaidIsland()
    local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    for i = 5, 1, -1 do
        for _, loc in ipairs(wOrigin.Locations:GetChildren()) do
            if loc.Name == ("Island " .. i) and (loc.Position - hrp.Position).Magnitude < 3000 then return loc end
        end
    end
end

local function IsInsideRaid()
    local t = PGui:FindFirstChild("Main") and PGui.Main:FindFirstChild("TopHUDList") and PGui.Main.TopHUDList:FindFirstChild("RaidTimer")
    return (t and t.Visible) or GetRaidIsland() ~= nil
end

local RaidLockUntil = 0

task.spawn(function()
    while getgenv().AutoRaid13k and task.wait(0.03) do
        pcall(function()
            local char = LP.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if not (hrp and hum and hum.Health > 0) then return end

            local frags = LP:FindFirstChild("Data") and LP.Data:FindFirstChild("Fragments") and LP.Data.Fragments.Value or 0
            local inRaid = IsInsideRaid()
            local raidIsl = GetRaidIsland()

            if frags >= 13000 and not inRaid then
                getgenv().IsFarmingRaidMob = false
                getgenv().AutoRaid13k = false
                StopTween()
                if hrp:FindFirstChild("FloatForce") then hrp.FloatForce:Destroy() end
                return
            end

            if inRaid and raidIsl then
                RaidLockUntil = tick() + 8
                local bestMob, minDist = nil, math.huge
                for _, mob in ipairs(workspace.Enemies:GetChildren()) do
                    local mHrp = mob:FindFirstChild("HumanoidRootPart")
                    local mHum = mob:FindFirstChildOfClass("Humanoid")
                    if mHrp and mHum and mHum.Health > 0 then
                        local d = (mHrp.Position - hrp.Position).Magnitude
                        if d < 1500 and d < minDist then bestMob, minDist = mob, d end
                    end
                end

                CurrentTargetMob = bestMob
                if bestMob then
                    EquipAndBuff()
                    BringMobs(bestMob)
                    -- Nếu máu thấp (<42%) -> Bay lên Y+220 hồi máu; nếu Boss tung skill -> Né Y+110; bình thường đứng Y+35
                    local offsetY = LowHealth and 220 or (tick() < DodgeUntil and 110 or 35)
                    getgenv().IsFarmingRaidMob = not LowHealth
                    TweenTo(bestMob.HumanoidRootPart.CFrame * CFrame.new(0, offsetY, 0), 260)
                else
                    getgenv().IsFarmingRaidMob = false
                    TweenTo(CFrame.new(raidIsl.Position + Vector3.new(0, 45, 0)), 260)
                end
                return
            end

            getgenv().IsFarmingRaidMob = false
            CurrentTargetMob = nil
            if tick() < RaidLockUntil then return end

            if not HasChip() then
                if not HasFruit() then
                    local f = GetCheapFruit()
                    if f then CommF:InvokeServer("LoadFruit", f) task.wait(0.7) else task.wait(2) return end
                end
                CommF:InvokeServer("RaidsNpc", "Select", "Flame")
                task.wait(0.8)
            end

            if HasChip() then
                local btn = workspace.Map:FindFirstChild("Boat Castle")
                    and workspace.Map["Boat Castle"]:FindFirstChild("RaidSummon2")
                    and workspace.Map["Boat Castle"].RaidSummon2:FindFirstChild("Button")
                    and workspace.Map["Boat Castle"].RaidSummon2.Button:FindFirstChild("Main")

                if not (btn and btn:FindFirstChild("ClickDetector")) then
                    ForceLoadPos(Vector3.new(-5034, 315, -2950))
                    task.wait(0.4)
                    return
                end

                hum:UnequipTools()
                fireclickdetector(btn.ClickDetector)
                task.wait(0.2)
                local chip = HasChip()
                if chip then
                    hum:EquipTool(chip)
                    task.wait(0.1)
                    fireclickdetector(btn.ClickDetector)
                end

                RaidLockUntil = tick() + 12
                task.wait(1)
                if HasChip() and not IsInsideRaid() and (hrp.Position - btn.Position).Magnitude > 20 then
                    TweenTo(btn.CFrame * CFrame.new(0, 0, 3), 200)
                    RaidLockUntil = tick() + 1.2
                end
            end
        end)
    end
end)
