local function GetFruitPrice(name)
	if not name then return 9999999 end
	if FruitPriceMap[name] then return FruitPriceMap[name] end
	local clean = name:gsub(" Fruit", "")
	for k, v in pairs(FruitPriceMap) do
		if string.find(k, clean) or string.find(name, k) then
			return v
		end
	end
	return 500000
end

-- Nhận diện CHÍNH XÁC trái ác quỷ vật lý (loại trừ skill trái đã ăn)
local function IsPhysicalFruit(t)
	if not (t and t:IsA("Tool")) then return false end
	if t.Name == "Special Microchip" then return false end
	if t:FindFirstChild("EatRemote") or t:FindFirstChild("EatFunction") then
		return true
	end
	if string.find(t.Name, "Fruit") and t.ToolTip ~= "Blox Fruit" then
		return true
	end
	return false
end

local function GetFruitInBag()
	local char = LP.Character
	for _, container in ipairs({ LP:FindFirstChild("Backpack"), char }) do
		if container then
			for _, t in ipairs(container:GetChildren()) do
				if IsPhysicalFruit(t) then
					return t
				end
			end
		end
	end
	return nil
end

local function JoinTeam(teamName)
	teamName = teamName or getgenv().Team or "Pirates"
	if LP.Team and LP.Team.Name == teamName and LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") then
		return true
	end

	local startTime = tick()
	while not Unloaded do
		if LP.Team and LP.Team.Name == teamName and LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") then
			break
		end
		if tick() - startTime > 10 then break end

		pcall(function() CommF:InvokeServer("SetTeam", teamName) end)
		pcall(function()
			local pg = LP:FindFirstChild("PlayerGui")
			local mainGui = pg and (pg:FindFirstChild("Main (minimal)") or pg:FindFirstChild("Main"))
			local choose = mainGui and mainGui:FindFirstChild("ChooseTeam", true)
			if choose and choose.Visible then
				local container = choose:FindFirstChild("Container") or choose
				local frame = container:FindFirstChild(teamName, true)
				local btn = frame and (frame:FindFirstChildWhichIsA("TextButton", true) or frame:FindFirstChildWhichIsA("ImageButton", true))
				if btn then
					if firesignal then
						firesignal(btn.Activated)
					elseif getconnections then
						for _, c in ipairs(getconnections(btn.Activated)) do c.Function() end
					end
					if GuiService and VIM then
						btn.Selectable = true
						GuiService.SelectedObject = btn
						VIM:SendKeyEvent(true, Enum.KeyCode.Return, false, game)
						task.wait(0.05)
						VIM:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
						GuiService.SelectedObject = nil
					end
				end
			end
		end)
		task.wait(0.4)
	end
	return (LP.Team and LP.Team.Name == teamName)
end

-- QUÉT KHO & MUA CHIP LIGHT
-- ====================================================================
local function ScanFruitInventory()
	local list, cheapKey, minPrice = {}, nil, 2000000

	-- 1. Nếu có trái vật lý sẵn trên người
	local physical = GetFruitInBag()
	if physical then
		local p = GetFruitPrice(physical.Name)
		local short = physical.Name:gsub(" Fruit", "")
		table.insert(list, "[" .. short .. " (Trên người)]")
		if p < minPrice then
			minPrice = p
			cheapKey = physical.Name
		end
	end

	-- 2. Quét kho bằng ItemReplicationService + ItemConfig
	pcall(function()
		local IRS = require(RS:WaitForChild("ItemReplicationService"))
		local IC = require(RS:WaitForChild("ItemConfig"))
		local KEYS = IRS.KEYS
		for _, item in IRS:GetItems(KEYS.QUANTITY) do
			if item.Value and item.Value > 0 then
				local ok, cfg = pcall(function() return IC.match(item.ItemId):unwrap() end)
				if ok and cfg and cfg.Display and cfg.Display.Category == "Blox Fruit" then
					local storageKey = (cfg.Index and cfg.Index.StorageKey) or cfg.Display.Name
					local price = GetFruitPrice(storageKey)
					if storageKey and price < 2000000 then
						local shortName = storageKey:gsub("%-.*", ""):gsub(" Fruit", "")
						table.insert(list, shortName .. " x" .. tostring(item.Value))
						if price < minPrice then
							minPrice = price
							cheapKey = storageKey
						end
					end
				end
			end
		end
	end)

	-- 3. Fallback getInventory
	if not cheapKey then
		pcall(function()
			local inv = CommF:InvokeServer("getInventory")
			if type(inv) == "table" then
				for _, item in pairs(inv) do
					if (item.Type == "Blox Fruit" or string.find(tostring(item.Name), "Fruit")) and item.Name then
						local name = item.Name
						local count = item.Count or item.Value or 1
						local price = GetFruitPrice(name)
						if price < 2000000 then
							local shortName = name:gsub("%-.*", ""):gsub(" Fruit", "")
							table.insert(list, shortName .. " x" .. tostring(count))
							if price < minPrice then
								minPrice = price
								cheapKey = name
							end
						end
					end
				end
			end
		end)
	end

	FruitText = "Fruit <2M: " .. (#list > 0 and table.concat(list, ", ") or "Trống")
	UpdateInfo()
	return cheapKey
end

local LastBuyAttempt = 0
local function BuyLightChip()
	if InRaid() or tick() < LoadRaidUntil or GetFragments() >= TargetFragments then
		return false
	end
	if HasChip() then return true end

	-- Chống spam mua chip liên tục
	if tick() - LastBuyAttempt < 3 then return false end
	LastBuyAttempt = tick()

	-- BƯỚC 1: Thử mua bằng Beli trước (100k Beli)
	SetStatus("Mua chip Light (Beli)...")
	pcall(function() CommF:InvokeServer("RaidsNpc", "Check") end)
	task.wait(0.2)
	pcall(function() CommF:InvokeServer("RaidsNpc", "Select", "Light") end)
	task.wait(0.5)
	if HasChip() then
		SetStatus("Đã mua chip (Beli)!")
		return true
	end

	-- BƯỚC 2: Kiểm tra trái vật lý trong người hoặc quét kho lấy ra
	local fruitInBag = GetFruitInBag()
	local cheapFruit = ScanFruitInventory()

	if not fruitInBag and cheapFruit then
		SetStatus("Lấy " .. cheapFruit .. " từ kho ra...")
		pcall(function() CommF:InvokeServer("LoadFruit", cheapFruit) end)
		task.wait(0.8)
		fruitInBag = GetFruitInBag()
	end

	-- BƯỚC 3: Đổi chip bằng trái vật lý (server RaidsNpc tự động nhận trái trong Backpack)
	if fruitInBag or cheapFruit then
		local fName = fruitInBag and fruitInBag.Name or cheapFruit
		SetStatus("Đổi " .. fName .. " lấy chip Light...")
		pcall(function() CommF:InvokeServer("RaidsNpc", "Check") end)
		task.wait(0.25)
		pcall(function() CommF:InvokeServer("RaidsNpc", "Select", "Light") end)
		task.wait(0.8)
		ScanFruitInventory()
	else
		SetStatus("Hết Beli & không có Fruit < 2M!")
		return false
	end

	return HasChip() ~= nil
end

local function RemoteStartRaid()
	if GetFragments() >= TargetFragments then return false end
	local cd = GetRaidButton()
	local char = LP.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local chip = HasChip()

	if cd and fireclickdetector then
		SetStatus("Bấm bệ Raid từ xa...")
		-- 1. Thử unequip tool
		if hum then pcall(function() hum:UnequipTools() end) end
		task.wait(0.15)
		fireclickdetector(cd)
		task.wait(0.15)

		-- 2. Equip Special Microchip
		chip = HasChip()
		if chip and hum then
			pcall(function() hum:EquipTool(chip) end)
			pcall(function() chip.Parent = char end)
			task.wait(0.2)
		end
		-- 3. Bấm lại ClickDetector
		fireclickdetector(cd)
		task.wait(0.8)

		if InRaid() or not HasChip() then
			return true
		end
	end

	-- Nếu sau khi bấm vẫn còn chip trên người (server check khoảng cách) hoặc chưa tìm thấy nút:
	-- Tự bay thẳng tới bệ bấm ở (-5034, 315, -2950)
	local isSea3 = (workspace:FindFirstChild("Map") and workspace.Map:FindFirstChild("Boat Castle")) or (game.PlaceId == 7449423635)
	local raidPos = isSea3 and Vector3.new(-5034, 315, -2950) or Vector3.new(-6438, 250, -4500)
	TargetPos = raidPos
	SetStatus("Bay tới bệ Raid...")
	return false
end

	if Gui and inst:IsDescendantOf(Gui) then return end

	local name = inst.Name
	if name == "RelicFire" or name == "RaidSummon2" or name == "Locations"
		or name == "Island 1" or name == "Island 2" or name == "Island 3" or name == "Island 4" or name == "Island 5" then
		return
	end

	pcall(function()
		local cls = inst.ClassName
		if cls == "ParticleEmitter" then
			if inst.Parent and inst.Parent.Name ~= "RelicFire" then
				inst.Lifetime, inst.Enabled = ZeroRange, false
			end
		elseif cls == "Trail" or cls == "Beam" then
			inst.Transparency, inst.Enabled = ZeroSequence, false
		elseif cls == "Smoke" or cls == "Fire" or cls == "Sparkles" then
			inst.Enabled = false
		elseif cls == "Explosion" then
			inst.Visible = false
		elseif cls == "Highlight" then
			inst.FillTransparency, inst.OutlineTransparency, inst.Enabled = 1, 1, false
		elseif cls == "SurfaceAppearance" then
			inst:Destroy()
		elseif LightTypes[cls] then
			inst.Enabled = false
		elseif cls == "Decal" or cls == "Texture" then
			inst.Transparency = 1
		elseif inst:IsA("MeshPart") then
			inst.CastShadow, inst.Reflectance, inst.Material = false, 0, Enum.Material.SmoothPlastic
			pcall(function() inst.TextureID = "" end)
		elseif inst:IsA("BasePart") then
			inst.CastShadow, inst.Reflectance = false, 0
			if inst.Material ~= Enum.Material.SmoothPlastic and inst.Material ~= Enum.Material.ForceField and inst.Material ~= Enum.Material.Neon then
				inst.Material = Enum.Material.SmoothPlastic
			end
		end
	end)
end

local function HideOtherCharacter(character)
	if not FpsBoostEnabled or not character or character == LP.Character then return end
	local function hide(d)
		pcall(function()
			if d:IsA("BasePart") then
				d.LocalTransparencyModifier, d.CastShadow = 1, false
			elseif d:IsA("Decal") or d:IsA("Texture") then
				d.Transparency = 1
			elseif d:IsA("ParticleEmitter") or d:IsA("Trail") or d:IsA("Beam") or d:IsA("Highlight") then
				d.Enabled = false
			end
		end)
	end
	for _, d in ipairs(character:GetDescendants()) do hide(d) end
	character.DescendantAdded:Connect(hide)
end

local FpsBoostApplied = false
local function ApplyFpsBoost()
	if not FpsBoostEnabled then return end
	LockFPS(FpsCap)
	pcall(function()
		workspace.InterpolationThrottling = Enum.InterpolationThrottlingMode.Enabled
	end)
	OptimizeLighting()
	OptimizeTerrain()
	OptimizeBloxFruits()

	if not FpsBoostApplied then
		FpsBoostApplied = true
		task.spawn(function()
			local list = workspace:GetDescendants()
			for i = 1, #list do
				if not FpsBoostEnabled then break end
				OptimizeInstance(list[i])
				if i % 300 == 0 then task.wait() end
			end
		end)
		table.insert(Conns, workspace.DescendantAdded:Connect(function(inst)
			if FpsBoostEnabled then task.defer(OptimizeInstance, inst) end
		end))
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= LP then
				if p.Character then task.spawn(HideOtherCharacter, p.Character) end
				table.insert(Conns, p.CharacterAdded:Connect(function(c) task.wait(0.5); HideOtherCharacter(c) end))
			end
		end
		table.insert(Conns, Players.PlayerAdded:Connect(function(p)
			if p ~= LP then
				table.insert(Conns, p.CharacterAdded:Connect(function(c) task.wait(0.5); HideOtherCharacter(c) end))
			end
		end))
	end
end

-- ====================================================================
local function CleanPhysics()
	TargetPos = nil
	CurrentEnemy = nil
	IsHealing = false
	local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
	if hrp and hrp:FindFirstChild("RaidFloat") then hrp.RaidFloat:Destroy() end
	if BringAnchor then pcall(function() BringAnchor:Destroy() end) BringAnchor = nil end
end

table.insert(Conns, RunService.Heartbeat:Connect(function(dt)
	local char = LP.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not Running or not hrp or not hum or hum.Health <= 0 then return end

	-- Chỉ tạo BodyVelocity khi đang có TargetPos hoặc đang trong Raid
	if TargetPos or InRaid() then
		if not hrp:FindFirstChild("RaidFloat") then
			local bv = Instance.new("BodyVelocity", hrp)
			bv.Name, bv.MaxForce, bv.Velocity = "RaidFloat", Vector3.new(1e6, 1e6, 1e6), Vector3.zero
		end
	else
		if hrp:FindFirstChild("RaidFloat") then
			hrp.RaidFloat:Destroy()
		end
	end

	if TargetPos then
		hum.Sit = false
		local clampedDt = math.min(dt, 0.05)
		local diff = TargetPos - hrp.Position
		local dist = diff.Magnitude
		local step = MoveSpeed * clampedDt
		local nextPos = dist <= step and TargetPos or (hrp.Position + diff.Unit * step)
		hrp.CFrame = CFrame.new(nextPos) * hrp.CFrame.Rotation
		hrp.AssemblyLinearVelocity = Vector3.zero
		hrp.AssemblyAngularVelocity = Vector3.zero
	end
end))

	end

-- FAST ATTACK LUỒNG RIÊNG (0.045s) & BRING MOB KHÔNG GIẬT
-- ====================================================================
local function IsBoss(mob)
	if not mob then return false end
	local name = mob.Name
	if string.find(name, "Master") or string.find(name, "Boss") or string.find(name, "Captain") then
		return true
	end
	local hum = mob:FindFirstChildOfClass("Humanoid")
	if hum and hum.MaxHealth >= 150000 then
		return true
	end
	return false
end

-- Bring Mob: neo quái thường, KHÔNG kéo Boss, freeze vị trí chống văng
local function BringMob(target)
	local tHrp = target and target:FindFirstChild("HumanoidRootPart")
	local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
	if not tHrp or not hrp or IsBoss(target) then return end

	if not BringAnchor or BringAnchor.Parent ~= workspace then
		if BringAnchor then pcall(function() BringAnchor:Destroy() end) end
		BringAnchor = Instance.new("Part", workspace)
		BringAnchor.Name = "BringAnchor"
		BringAnchor.Anchored = true
		BringAnchor.CanCollide = false
		BringAnchor.Transparency = 1
		BringAnchor.Size = Vector3.one
	end
	BringAnchor.CFrame = tHrp.CFrame

	for _, e in ipairs(workspace.Enemies:GetChildren()) do
		local eHrp, eHum = e:FindFirstChild("HumanoidRootPart"), e:FindFirstChildOfClass("Humanoid")
		if e ~= target and eHrp and eHum and eHum.Health > 0 and not IsBoss(e) and not e:FindFirstChild("Ignored") then
			if (eHrp.Position - BringAnchor.Position).Magnitude <= 350 and (not isnetworkowner or isnetworkowner(eHrp)) then
				for _, p in ipairs(e:GetChildren()) do
					if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false end
				end
				eHum.PlatformStand = true
				eHum.WalkSpeed = 0
				eHrp.AssemblyLinearVelocity = Vector3.zero
				eHrp.AssemblyAngularVelocity = Vector3.zero
				eHrp.CFrame = BringAnchor.CFrame * CFrame.new(math.random(-1, 1), 0, math.random(-1, 1))

				if not e:GetAttribute("BringWatch") then
					e:SetAttribute("BringWatch", true)
					task.spawn(function()
						local hp = eHum.Health
						task.wait(3.5)
						if e and e.Parent and eHum.Health == hp and eHum.Health > 0 then
							eHrp.CFrame = e.WorldPivot
							Instance.new("IntValue", e).Name = "Ignored"
						end
					end)
				end
			end
		end
	end
end

-- Luồng Fast Attack Melee độc lập
task.spawn(function()
	local regHit = nil
	pcall(function()
		regHit = require(RS.Modules.Net):RemoteEvent("RegisterHit", true)
	end)

	while not Unloaded do
		task.wait(0.045)
		if not Running or not InRaid() or not CurrentEnemy then continue end

		local char = LP.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if not hrp or not hum or hum.Health <= 0 then continue end

		-- Tự bật Buso
		if not char:FindFirstChild("HasBuso") then
			pcall(function() CommF:InvokeServer("Buso") end)
		end

		-- Tự trang bị Melee
		local tool = char:FindFirstChildOfClass("Tool")
		if not tool or tool.ToolTip ~= "Melee" then
			for _, t in ipairs(LP.Backpack:GetChildren()) do
				if t:IsA("Tool") and t.ToolTip == "Melee" then
					hum:EquipTool(t)
					tool = t
					break
				end
			end
		end

		local hits = {}
		for _, e in ipairs(workspace.Enemies:GetChildren()) do
			local eHrp, eHum = e:FindFirstChild("HumanoidRootPart"), e:FindFirstChildOfClass("Humanoid")
			if eHrp and eHum and eHum.Health > 0 and (eHrp.Position - hrp.Position).Magnitude <= 65 then
				table.insert(hits, {e, e:FindFirstChild("Head") or eHrp})
			end
		end

		if #hits > 0 then
			pcall(function()
				Net["RE/RegisterAttack"]:FireServer(0)
				if regHit then
					regHit:FireServer(hits[1][2], hits)
				elseif Net:FindFirstChild("RE/RegisterHit") then
					Net["RE/RegisterHit"]:FireServer(hits[1][2], hits)
				end
			end)
		end
	end
end)

ScanFruitInventory()

task.spawn(function()
	while not Unloaded do
		task.wait(0.1)
		local ok, err = pcall(function()
			-- 1. Kiểm tra đủ 14k fragments
			if GetFragments() >= TargetFragments and not InRaid() then
				Running = false
				SetStatus("DONE (Đã đủ 14k F!)")
				Status.TextColor3 = Color3.fromRGB(80, 255, 120)
				ToggleBtn.Text = "DONE"
				ToggleBtn.BackgroundColor3 = Color3.fromRGB(40, 140, 60)
				CleanPhysics()
				UpdateInfo()
				task.wait(1)
				return
			end

			if not Running then return end

			-- Đảm bảo đã chọn team
			if not LP.Team then JoinTeam(getgenv().Team) end

			local char = LP.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if not hrp or not hum then return end

			local inRaidNow = InRaid()

			-- Cập nhật kho sau khi kết thúc lượt Raid (không spam 3s)
			if WasInRaid and not inRaidNow then
				SetStatus("Raid xong! Cập nhật kho...")
				task.wait(2)
				ScanFruitInventory()
				UpdateInfo()
			end
			WasInRaid = inRaidNow

			-- 2. Logic trong Raid
			if inRaidNow then
				LoadRaidUntil = 0
				pcall(sethiddenproperty, LP, "SimulationRadius", math.huge)

				-- Tìm quái gần nhất
				local target, minDist = nil, 2500
				for _, e in ipairs(workspace.Enemies:GetChildren()) do
					local eHrp, eHum = e:FindFirstChild("HumanoidRootPart"), e:FindFirstChildOfClass("Humanoid")
					if eHrp and eHum and eHum.Health > 0 then
						local d = (eHrp.Position - hrp.Position).Magnitude
						if d < minDist then minDist, target = d, e end
					end
				end

				CurrentEnemy = target

				if target then
					local tHrp = target:FindFirstChild("HumanoidRootPart")
					local tHum = target:FindFirstChildOfClass("Humanoid")
					local targetIsBoss = IsBoss(target)

					-- Cơ chế chống chết Boss đảo 5
					local hpPercent = hum.Health / hum.MaxHealth
					if hpPercent < 0.40 then
						IsHealing = true
					elseif hpPercent >= 0.75 then
						IsHealing = false
					end

					if IsHealing and tHrp then
						-- Máu dưới 40%: Bay lên Y+180 để né skill và hồi máu
						SetStatus(string.format("Hồi máu (%d%%)...", math.floor(hpPercent * 100)))
						TargetPos = tHrp.Position + Vector3.new(0, 180, 0)
					elseif targetIsBoss and tHrp then
						-- Đánh Boss: Lượn vòng tròn trên cao (Y+30, bán kính 20) quanh Boss
						SetStatus("Đánh Boss: " .. target.Name)
						local angle = tick() * 2.5
						local offsetX = math.cos(angle) * 20
						local offsetZ = math.sin(angle) * 20
						TargetPos = tHrp.Position + Vector3.new(offsetX, 30, offsetZ)
					else
						-- Đánh quái thường: Neo vị trí và gom quái
						SetStatus("Đang đánh: " .. target.Name)
						if tHrp then
							TargetPos = tHrp.Position + Vector3.new(0, 24, 0)
							if (tHrp.Position - hrp.Position).Magnitude <= 65 then
								BringMob(target)
							end
						end
					end
				else
					-- Tìm đảo tiếp theo
					local locs = (workspace:FindFirstChild("_WorldOrigin") and workspace._WorldOrigin:FindFirstChild("Locations")) or workspace:FindFirstChild("Locations")
					local foundIsland = false
					if locs then
						for i = 5, 1, -1 do
							local isl = locs:FindFirstChild("Island " .. i)
							if isl and (isl.Position - hrp.Position).Magnitude <= 6000 then
								SetStatus("Qua đảo " .. i .. "...")
								TargetPos = isl.Position + Vector3.new(0, 70, 0)
								foundIsland = true
								break
							end
						end
					end
					if not foundIsland then
						SetStatus("Đợi đảo tiếp theo...")
					end
				end
			else
				-- 3. Logic ngoài Raid
				CurrentEnemy = nil
				IsHealing = false

				if HasChip() then
					if RemoteStartRaid() then
						LoadRaidUntil = tick() + 25
						task.wait(1)
					end
				elseif tick() < LoadRaidUntil then
					SetStatus("Đang load vào Raid...")
				elseif not BuyLightChip() then
					task.wait(2)
				end
			end
		end)

		-- Hiển thị lỗi trực tiếp lên UI Status nếu có lỗi trong Main Loop
		if not ok and err then
			local errStr = tostring(err)
			local cleanErr = errStr:match(":(%d+: .*)") or errStr:sub(1, 40)
			SetStatus("Lỗi: " .. cleanErr)
			warn("[KaitunLightRaid Error]:", err)
		end
	end
end)
