local function JoinTeam(teamName)
	teamName = teamName or getgenv().Team or "Pirates"
	if LP.Team and LP.Team.Name == teamName and LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") then
		return true
	end

	local startTime = tick()
	while not Unloaded do
		if LP.Team and LP.Team.Name == teamName and LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") then
			break
		end
		if tick() - startTime > 10 then break end

		pcall(function() CommF:InvokeServer("SetTeam", teamName) end)

		pcall(function()
			local pg = LP:FindFirstChild("PlayerGui")
			local mainGui = pg and (pg:FindFirstChild("Main (minimal)") or pg:FindFirstChild("Main"))
			local choose = mainGui and mainGui:FindFirstChild("ChooseTeam", true)
			if choose and choose.Visible then
				local container = choose:FindFirstChild("Container") or choose
				local frame = container:FindFirstChild(teamName, true)
				local btn = frame and (frame:FindFirstChildWhichIsA("TextButton", true) or frame:FindFirstChildWhichIsA("ImageButton", true))
				if btn then
					if firesignal then
						firesignal(btn.Activated)
					elseif getconnections then
						for _, c in ipairs(getconnections(btn.Activated)) do c.Function() end
					end
					if GuiService and VIM then
						btn.Selectable = true
						GuiService.SelectedObject = btn
						VIM:SendKeyEvent(true, Enum.KeyCode.Return, false, game)
						task.wait(0.05)
						VIM:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
						GuiService.SelectedObject = nil
					end
				end
			end
		end)
		task.wait(0.4)
	end
	return (LP.Team and LP.Team.Name == teamName)
end

local ZeroRange = NumberRange.new(0)
local ZeroSequence = NumberSequence.new(1)
local LightTypes = { PointLight = true, SpotLight = true, SurfaceLight = true }

local function OptimizeLighting()
	pcall(function()
		Lighting.GlobalShadows = false
		Lighting.FogEnd = 9e9
		Lighting.Brightness = 0
		Lighting.Ambient = Color3.fromRGB(128, 128, 128)
		Lighting.OutdoorAmbient = Color3.fromRGB(128, 128, 128)
	end)
	for _, v in ipairs(Lighting:GetDescendants()) do
		pcall(function()
			if v:IsA("PostEffect") or v:IsA("Atmosphere") or v:IsA("Clouds") then v.Enabled = false end
		end)
	end
	if sethiddenproperty then pcall(sethiddenproperty, Lighting, "Technology", Enum.Technology.Compatibility) end
end

local function OptimizeTerrain()
	local terrain = workspace:FindFirstChildOfClass("Terrain")
	if terrain then
		pcall(function()
			terrain.WaterWaveSize = 0
			terrain.WaterWaveSpeed = 0
			terrain.WaterReflectance = 0
			terrain.Decoration = false
		end)
	end
end

local function OptimizeInstance(inst)
	if not FpsBoostEnabled or not inst or not inst.Parent then return end
	local char = LP.Character
	if char and (inst == char or inst:IsDescendantOf(char)) then return end
	if Gui and inst:IsDescendantOf(Gui) then return end

	local name = inst.Name
	if name == "RelicFire" or name == "RaidSummon2" or name == "Locations"
		or name == "Island 1" or name == "Island 2" or name == "Island 3" or name == "Island 4" or name == "Island 5" then
		return
	end

	pcall(function()
		local cls = inst.ClassName
		if cls == "ParticleEmitter" then
			if inst.Parent and inst.Parent.Name ~= "RelicFire" then
				inst.Lifetime, inst.Enabled = ZeroRange, false
			end
		elseif cls == "Trail" or cls == "Beam" then
			inst.Transparency, inst.Enabled = ZeroSequence, false
		elseif cls == "Smoke" or cls == "Fire" or cls == "Sparkles" then
			inst.Enabled = false
		elseif cls == "Explosion" then
			inst.Visible = false
		elseif cls == "Highlight" then
			inst.FillTransparency, inst.OutlineTransparency, inst.Enabled = 1, 1, false
		elseif cls == "SurfaceAppearance" then
			inst:Destroy()
		elseif LightTypes[cls] then
			inst.Enabled = false
		elseif cls == "Decal" or cls == "Texture" then
			inst.Transparency = 1
		elseif inst:IsA("MeshPart") then
			inst.CastShadow, inst.Reflectance, inst.Material = false, 0, Enum.Material.SmoothPlastic
			pcall(function() inst.TextureID = "" end)
		elseif inst:IsA("BasePart") then
			inst.CastShadow, inst.Reflectance = false, 0
			if inst.Material ~= Enum.Material.SmoothPlastic and inst.Material ~= Enum.Material.ForceField and inst.Material ~= Enum.Material.Neon then
				inst.Material = Enum.Material.SmoothPlastic
			end
		end
	end)
end

local function HideOtherCharacter(character)
	if not FpsBoostEnabled or not character or character == LP.Character then return end
	local function hide(d)
		pcall(function()
			if d:IsA("BasePart") then
				d.LocalTransparencyModifier, d.CastShadow = 1, false
			elseif d:IsA("Decal") or d:IsA("Texture") then
				d.Transparency = 1
			elseif d:IsA("ParticleEmitter") or d:IsA("Trail") or d:IsA("Beam") or d:IsA("Highlight") then
				d.Enabled = false
			end
		end)
	end
	for _, d in ipairs(character:GetDescendants()) do hide(d) end
	character.DescendantAdded:Connect(hide)
end

local FpsBoostApplied = false
local function ApplyFpsBoost()
	if not FpsBoostEnabled then return end
	LockFPS(FpsCap)
	pcall(function()
		settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
		workspace.InterpolationThrottling = Enum.InterpolationThrottlingMode.Enabled
	end)
	OptimizeLighting()
	OptimizeTerrain()
	OptimizeBloxFruits()

	if not FpsBoostApplied then
		FpsBoostApplied = true
		task.spawn(function()
			local list = workspace:GetDescendants()
			for i = 1, #list do
				if not FpsBoostEnabled then break end
				OptimizeInstance(list[i])
				if i % 300 == 0 then task.wait() end
			end
		end)
		table.insert(Conns, workspace.DescendantAdded:Connect(function(inst)
			if FpsBoostEnabled then task.defer(OptimizeInstance, inst) end
		end))
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= LP then
				if p.Character then task.spawn(HideOtherCharacter, p.Character) end
				table.insert(Conns, p.CharacterAdded:Connect(function(c) task.wait(0.5); HideOtherCharacter(c) end))
			end
		end
		table.insert(Conns, Players.PlayerAdded:Connect(function(p)
			if p ~= LP then
				table.insert(Conns, p.CharacterAdded:Connect(function(c) task.wait(0.5); HideOtherCharacter(c) end))
			end
		end))
	end
end

-- Quét trái dưới 2M Beli
local function ScanFruitInventory()
	local list, cheapKey, minPrice = {}, nil, 2000000
	pcall(function()
		local IRS = require(RS.ItemReplicationService)
		local MATCH = require(RS.ItemConfig.Storage).match
		for _, v in ipairs(IRS:GetItems(IRS.KEYS.QUANTITY)) do
			if v.Value and v.Value > 0 then
				local item = MATCH(v.ItemId)._ok
				if item and item.Display.Category == "Blox Fruit" and item.Quality.MoneyPrice < 2000000 then
					local key = item.Index.StorageKey
					local shortName = (item.Display.Name or key):gsub(" Fruit", "")
					table.insert(list, shortName .. " x" .. v.Value)
					if item.Quality.MoneyPrice < minPrice then
						minPrice = item.Quality.MoneyPrice
						cheapKey = key
					end
				end
			end
		end
	end)
	FruitText = "Fruit <2M: " .. (#list > 0 and table.concat(list, ", ") or "Trống")
	UpdateInfo()
	return cheapKey
end

local function BuyLightChip()
	if InRaid() or tick() < LoadRaidUntil or GetFragments() >= TargetFragments then return false end
	local cheapFruit = ScanFruitInventory()
	if HasChip() then return true end

	SetStatus("Mua chip Light (Beli)...")
	CommF:InvokeServer("RaidsNpc", "Select", "Light")
	task.wait(0.3)
	if HasChip() then return true end

	if cheapFruit then
		SetStatus("Đổi " .. cheapFruit .. " lấy chip Light...")
		CommF:InvokeServer("LoadFruit", cheapFruit)
		task.wait(0.35)
		CommF:InvokeServer("RaidsNpc", "Select", "Light")
		task.wait(0.3)
		ScanFruitInventory()
	end
	return HasChip() ~= nil
end

	local map = workspace.Map

	if cd and fireclickdetector then
		fireclickdetector(cd)

	end
	return false
end

local function BringMob(mob)
	local mHrp = mob and mob:FindFirstChild("HumanoidRootPart")
	local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
	if not mHrp or not hrp then return end

	if BringTarget ~= mob then
		BringTarget, BringCF = mob, mHrp.CFrame
		for _, e in ipairs(workspace.Enemies:GetChildren()) do
			if e:FindFirstChild("Ignored") then e.Ignored:Destroy() end
		end
	end
	if (hrp.Position - mHrp.Position).Magnitude > 60 then return end

	for _, e in ipairs(workspace.Enemies:GetChildren()) do
		local eHrp, eHum = e:FindFirstChild("HumanoidRootPart"), e:FindFirstChildOfClass("Humanoid")
		if eHrp and eHum and eHum.Health > 0 and not e:FindFirstChild("Ignored") then
			if (eHrp.Position - BringCF.Position).Magnitude <= 350 and (not isnetworkowner or isnetworkowner(eHrp)) then
				for _, p in ipairs(e:GetChildren()) do
					if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false end
				end
				eHrp.CFrame = BringCF * CFrame.new(0, math.random(0, 2), math.random(0, 2))
				if not e:GetAttribute("BringWatch") then
					e:SetAttribute("BringWatch", true)
					task.spawn(function()
						local hp = eHum.Health
						task.wait(3.5)
						if e and e.Parent and eHum.Health == hp and eHum.Health > 0 and not e:FindFirstChild("Ignored") then
							eHrp.CFrame = e.WorldPivot
							Instance.new("IntValue", e).Name = "Ignored"
						end
					end)
				end
			end
		end
	end
end

-- Tự trang bị và đánh bằng Melee
local function AttackMelee()
	local char = LP.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum then return end

	if not char:FindFirstChild("HasBuso") then
		pcall(function() CommF:InvokeServer("Buso") end)
	end

	local tool = char:FindFirstChildOfClass("Tool")
	if not tool or tool.ToolTip ~= "Melee" then
		for _, t in ipairs(LP.Backpack:GetChildren()) do
			if t:IsA("Tool") and t.ToolTip == "Melee" then
				hum:EquipTool(t)
				tool = t
				break
			end
		end
	end
	if not tool or tool.ToolTip ~= "Melee" then return end

	local hits = {}
	for _, e in ipairs(workspace.Enemies:GetChildren()) do
		local eHrp, eHum = e:FindFirstChild("HumanoidRootPart"), e:FindFirstChildOfClass("Humanoid")
		if eHrp and eHum and eHum.Health > 0 and (eHrp.Position - hrp.Position).Magnitude <= 65 then
			table.insert(hits, {e, e:FindFirstChild("Head") or eHrp})
		end
	end
	if #hits > 0 then
		Net["RE/RegisterAttack"]:FireServer(0)
		Net["RE/RegisterHit"]:FireServer(hits[1][2], hits)
	end
end

ScanFruitInventory()

task.spawn(function()
	local lastScan = 0
	while not Unloaded do
		task.wait(0.1)
		pcall(function()
			-- Kiểm tra điều kiện hoàn thành 14,000 Fragments
			if GetFragments() >= TargetFragments and not InRaid() then
				Running = false
				SetStatus("DONE (Đã đủ 14k F!)")
				Status.TextColor3 = Color3.fromRGB(80, 255, 120)
				ToggleBtn.Text = "DONE"
				ToggleBtn.BackgroundColor3 = Color3.fromRGB(40, 140, 60)
				CleanPhysics()
				UpdateInfo()
				task.wait(1)
				return
			end

			if not Running then return end

			if not LP.Team then JoinTeam(getgenv().Team) end

			if tick() - lastScan >= 3 then
				lastScan = tick()
				ScanFruitInventory()
			end

			local char = LP.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if not hrp then return end

			if InRaid() then
				LoadRaidUntil = 0
				pcall(sethiddenproperty, LP, "SimulationRadius", math.huge)
				local target, minDist = nil, 2500
				for _, e in ipairs(workspace.Enemies:GetChildren()) do
					local eHrp, eHum = e:FindFirstChild("HumanoidRootPart"), e:FindFirstChildOfClass("Humanoid")
					if eHrp and eHum and eHum.Health > 0 then
						local d = (eHrp.Position - hrp.Position).Magnitude
						if d < minDist then minDist, target = d, e end
					end
				end

				if target then
					SetStatus("Đang đánh: " .. target.Name)
					TargetPos = target.HumanoidRootPart.Position + Vector3.new(0, 26, 8)
					if (target.HumanoidRootPart.Position - hrp.Position).Magnitude <= 60 then
						BringMob(target)
						AttackMelee()
					end
				else
					for i = 5, 1, -1 do
						local isl = workspace._WorldOrigin.Locations:FindFirstChild("Island " .. i)
						if isl and (isl.Position - hrp.Position).Magnitude <= 6000 then
							SetStatus("Qua đảo " .. i .. "...")
							TargetPos = isl.Position + Vector3.new(0, 70, 0)
							break
						end
					end
				end
			else
				TargetPos = nil
				if HasChip() then
					if RemoteStartRaid() then
						LoadRaidUntil = tick() + 18
						task.wait(1)
					end
				elseif tick() < LoadRaidUntil then
					SetStatus("Đang load vào Raid...")
				elseif not BuyLightChip() then
					SetStatus("Hết Beli & Fruit dưới 2M!")
					task.wait(3)
				end
			end
		end)
	end
end)
