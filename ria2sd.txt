local CurrentTween = nil
local function StopTween()
    if CurrentTween then pcall(function() CurrentTween:Cancel() end) CurrentTween = nil end
end

RunService.Stepped:Connect(function()
    if not getgenv().AutoRaid13k then return end
    pcall(function()
        if PGui:FindFirstChild("Main") and PGui.Main:FindFirstChild("DamageCounter") then
            PGui.Main.DamageCounter.Enabled = false
        end
        if PGui:FindFirstChild("Notifications") then
            PGui.Notifications.Enabled = false
        end
        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not (hrp and hum and hum.Health > 0) then return end
        if hum.Sit then hum.Sit = false end

        if not hrp:FindFirstChild("FloatForce") then
            local bv = Instance.new("BodyVelocity", hrp)
            bv.Name, bv.Velocity, bv.MaxForce, bv.P = "FloatForce", Vector3.zero, Vector3.new(1e5, 1e5, 1e5), 10000
        end
        hrp.AssemblyLinearVelocity = Vector3.zero
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") and part.CanCollide then part.CanCollide = false end
        end
    end)
end)

local function TweenTo(targetCF)
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not (hrp and hum and hum.Health > 0) then return end

    local dist = (hrp.Position - targetCF.Position).Magnitude
    if dist <= 12 then
        StopTween()
        hrp.CFrame = targetCF
        return
    end
    StopTween()
    if dist > 60 then
        hrp.CFrame = CFrame.new(hrp.Position.X, targetCF.Position.Y, hrp.Position.Z)
    end
    CurrentTween = TS:Create(hrp, TweenInfo.new(dist / 200, Enum.EasingStyle.Linear), { CFrame = targetCF })
    CurrentTween:Play()
end

local FruitPrices = {}
local function GetCheapFruit()
    if not next(FruitPrices) then
        local ok, data = pcall(function() return CommF:InvokeServer("GetFruits", false) end)
        if ok and type(data) == "table" then
            for _, f in pairs(data) do if f.Name and f.Price then FruitPrices[f.Name] = tonumber(f.Price) end end
        end
    end
    local best, minP = nil, math.huge
    pcall(function()
        local IRS = require(RS:WaitForChild("ItemReplicationService"))
        local IC = require(RS:WaitForChild("ItemConfig"))
        for _, item in IRS:GetItems(IRS.KEYS.QUANTITY) do
            if item.Value and item.Value > 0 then
                local ok, cfg = pcall(function() return IC.match(item.ItemId):unwrap() end)
                if ok and cfg and cfg.Display and cfg.Display.Category == "Blox Fruit" then
                    local key = (cfg.Index and cfg.Index.StorageKey) or cfg.Display.Name
                    local p = FruitPrices[key] or tonumber(cfg.Price) or 500000
                    if key and p < 1500000 and p < minP then best, minP = key, p end
                end
            end
        end
    end)
    if not best then
        pcall(function()
            for _, v in pairs(CommF:InvokeServer("getInventory") or {}) do
                if (v.Type == "Blox Fruit" or string.find(tostring(v.Name), "-")) and v.Name then
                    local p = tonumber(v.Value) or FruitPrices[v.Name] or 500000
                    if p < 1500000 and p < minP then best, minP = v.Name, p end
                end
            end
        end)
    end
    return best
end

local function HasChip()
    return LP.Backpack:FindFirstChild("Special Microchip") or (LP.Character and LP.Character:FindFirstChild("Special Microchip"))
end

local function HasFruit()
    for _, c in ipairs({ LP.Backpack, LP.Character }) do
        if c then
            for _, t in ipairs(c:GetChildren()) do
                if t:IsA("Tool") and (string.find(t.Name, "Fruit") or t:FindFirstChild("EatRemote")) then return t end
            end
        end
    end
end

local CombatUtil = require(RS.Modules.CombatUtil)
local Net = require(RS.Modules.Net)
local RegAttack = RS.Modules.Net:WaitForChild("RE/RegisterAttack")
local RegHit = Net:RemoteEvent("RegisterHit", true)

task.spawn(function()
    while task.wait(0.04) do
        if not getgenv().AutoRaid13k then break end
        if getgenv().IsFarmingRaidMob then
            pcall(function()
                local char = LP.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                if not hrp or (char:FindFirstChild("Stun") and char.Stun.Value ~= 0) then return end

                local hits = {}
                for _, mob in ipairs(workspace.Enemies:GetChildren()) do
                    local mHrp = mob:FindFirstChild("HumanoidRootPart")
                    local mHum = mob:FindFirstChildOfClass("Humanoid")
                    if mHrp and mHum and mHum.Health > 0 and (mHrp.Position - hrp.Position).Magnitude <= 75 then
                        local hitPart = mob:FindFirstChild("Head") or mHrp
                        if not CombatUtil or CombatUtil:IsVulnerable(mob) then
                            table.insert(hits, { mob, hitPart })
                        end
                    end
                end
                if #hits > 0 then
                    local firstPart = table.remove(hits, 1)[2]
                    RegAttack:FireServer(0)
                    RegHit:FireServer(firstPart, hits)
                end
            end)
        end
    end
end)

local function BringMobsToPoint(pullPos)
    pcall(sethiddenproperty, LP, "SimulationRadius", math.huge)
    for _, mob in ipairs(workspace.Enemies:GetChildren()) do
        local mHrp = mob:FindFirstChild("HumanoidRootPart")
        local mHum = mob:FindFirstChildOfClass("Humanoid")
        if mHrp and mHum and mHum.Health > 0 and (mHrp.Position - pullPos).Magnitude <= 350 then
            if not string.find(mob.Name, "Master") and mHum.MaxHealth < 45000 then
                pcall(function()
                    for _, p in ipairs(mob:GetChildren()) do
                        if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false end
                    end
                    mHum.WalkSpeed = 0
                    mHum.JumpPower = 0
                    mHum.AutoRotate = false
                    mHum.PlatformStand = true
                    mHrp.AssemblyLinearVelocity = Vector3.zero
                    mHrp.AssemblyAngularVelocity = Vector3.zero

                    local att = mHrp:FindFirstChild("dutditattach")
                    if not att then
                        att = Instance.new("Attachment", mHrp)
                        att.Name = "dutditattach"
                        local ap = Instance.new("AlignPosition", att)
                        ap.Name = "dutditaglin"
                        ap.Attachment0 = att
                        ap.Mode = Enum.PositionAlignmentMode.OneAttachment
                        ap.MaxForce = 1000000
                        ap.MaxVelocity = 350
                        ap.Responsiveness = 40
                        local ao = Instance.new("AlignOrientation", att)
                        ao.Name = "dutditorient"
                        ao.Attachment0 = att
                        ao.Mode = Enum.OrientationAlignmentMode.OneAttachment
                        ao.MaxTorque = 1000000
                        ao.Responsiveness = 40
                    end
                    local ap = att:FindFirstChild("dutditaglin")
                    if ap then ap.Position = pullPos end
                    mHrp.CFrame = CFrame.new(pullPos)
                end)
            end
        end
    end
end

local LastBuff = 0
local function EquipAndBuff()
    local char = LP.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not (char and hum and hum.Health > 0) then return end

    local tool = char:FindFirstChildOfClass("Tool")
    if not (tool and tool.ToolTip == "Melee") then
        for _, t in ipairs(LP.Backpack:GetChildren()) do
            if t:IsA("Tool") and t.ToolTip == "Melee" then hum:EquipTool(t) break end
        end
    end

    if tick() - LastBuff > 2 then
        LastBuff = tick()
        if not char:FindFirstChild("HasBuso") then pcall(function() CommF:InvokeServer("Buso") end) end
        pcall(function() CommE:FireServer("ActivateAbility") end)
        local awk = LP.Backpack:FindFirstChild("Awakening") or char:FindFirstChild("Awakening")
        if awk and awk:FindFirstChild("RemoteFunction") then
            pcall(function() awk.RemoteFunction:InvokeServer(true) end)
        end
    end
end

local function GetRaidIsland()
    local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    for i = 5, 1, -1 do
        for _, loc in ipairs(wOrigin.Locations:GetChildren()) do
            if loc.Name == ("Island " .. i) and (loc.Position - hrp.Position).Magnitude < 3000 then return loc end
        end
    end
end

local function IsInsideRaid()
    local t = PGui:FindFirstChild("Main") and PGui.Main:FindFirstChild("TopHUDList") and PGui.Main.TopHUDList:FindFirstChild("RaidTimer")
    return (t and t.Visible) or GetRaidIsland() ~= nil
end

local RaidLockUntil = 0
local AnchorPos = nil
local OrbitAngle = 0

task.spawn(function()
    while getgenv().AutoRaid13k and task.wait(0.05) do
        pcall(function()
            local char = LP.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if not (hrp and hum and hum.Health > 0) then return end

            local frags = LP:FindFirstChild("Data") and LP.Data:FindFirstChild("Fragments") and LP.Data.Fragments.Value or 0
            local inRaid = IsInsideRaid()
            local raidIsl = GetRaidIsland()

            if frags >= 13000 and not inRaid then
                getgenv().IsFarmingRaidMob = false
                getgenv().AutoRaid13k = false
                StopTween()
                if hrp:FindFirstChild("FloatForce") then hrp.FloatForce:Destroy() end
                return
            end

            if inRaid and raidIsl then
                RaidLockUntil = tick() + 8
                local bossMob, nearestMob, minDist = nil, nil, math.huge

                for _, mob in ipairs(workspace.Enemies:GetChildren()) do
                    local mHrp = mob:FindFirstChild("HumanoidRootPart")
                    local mHum = mob:FindFirstChildOfClass("Humanoid")
                    if mHrp and mHum and mHum.Health > 0 then
                        local d = (mHrp.Position - raidIsl.Position).Magnitude
                        if d < 1500 then
                            if string.find(mob.Name, "Master") or mHum.MaxHealth >= 45000 then
                                bossMob = mob
                            end
                            local distToPlr = (mHrp.Position - hrp.Position).Magnitude
                            if distToPlr < minDist then
                                nearestMob, minDist = mob, distToPlr
                            end
                        end
                    end
                end

                local target = bossMob or nearestMob
                if target then
                    getgenv().IsFarmingRaidMob = true
                    EquipAndBuff()

                    if bossMob then
                        AnchorPos = bossMob.HumanoidRootPart.Position
                        BringMobsToPoint(AnchorPos)
                        OrbitAngle = (OrbitAngle + 35) % 360
                        local rad = math.rad(OrbitAngle)
                        local standPos = AnchorPos + Vector3.new(math.cos(rad) * 18, 28, math.sin(rad) * 18)
                        TweenTo(CFrame.new(standPos))
                    else
                        local hasMobNearAnchor = false
                        if AnchorPos then
                            for _, mob in ipairs(workspace.Enemies:GetChildren()) do
                                local mHrp = mob:FindFirstChild("HumanoidRootPart")
                                local mHum = mob:FindFirstChildOfClass("Humanoid")
                                if mHrp and mHum and mHum.Health > 0 and (mHrp.Position - AnchorPos).Magnitude <= 320 then
                                    hasMobNearAnchor = true
                                    break
                                end
                            end
                        end
                        if not hasMobNearAnchor then
                            AnchorPos = target.HumanoidRootPart.Position
                        end
                        BringMobsToPoint(AnchorPos)
                        TweenTo(CFrame.new(AnchorPos + Vector3.new(0, 22, 0)))
                    end
                else
                    getgenv().IsFarmingRaidMob = false
                    AnchorPos = nil
                    TweenTo(CFrame.new(raidIsl.Position + Vector3.new(0, 50, 0)))
                end
                return
            end

            getgenv().IsFarmingRaidMob = false
            AnchorPos = nil
            if tick() < RaidLockUntil then return end

            if not HasChip() then
                if not HasFruit() then
                    local f = GetCheapFruit()
                    if f then CommF:InvokeServer("LoadFruit", f) task.wait(0.8) else task.wait(2) return end
                end
                CommF:InvokeServer("RaidsNpc", "Select", "Flame")
                task.wait(1)
            end

            if HasChip() then
                local boatCastle = workspace.Map:FindFirstChild("Boat Castle") or workspace:FindFirstChild("Boat Castle")
                local btn = boatCastle and boatCastle:FindFirstChild("RaidSummon2")
                    and boatCastle.RaidSummon2:FindFirstChild("Button")
                    and boatCastle.RaidSummon2.Button:FindFirstChild("Main")

                if not (btn and btn:FindFirstChild("ClickDetector")) then
                    TweenTo(CFrame.new(-5034, 315, -2950))
                    return
                end

                hum:UnequipTools()
                fireclickdetector(btn.ClickDetector)
                task.wait(0.2)
                local chip = HasChip()
                if chip then
                    hum:EquipTool(chip)
                    task.wait(0.15)
                    fireclickdetector(btn.ClickDetector)
                end

                RaidLockUntil = tick() + 12
                task.wait(1.2)
                if HasChip() and not IsInsideRaid() and (hrp.Position - btn.Position).Magnitude > 20 then
                    TweenTo(btn.CFrame * CFrame.new(0, 0, 3))
                    RaidLockUntil = tick() + 1.5
                end
            end
        end)
    end
end)
